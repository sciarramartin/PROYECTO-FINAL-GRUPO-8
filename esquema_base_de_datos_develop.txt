    carreras {

        string nombre

    }

        datetime updatedAt

        string estado
    }

    amistades {
        int id PK

        int id_usuario_destino FK
        string estado
        datetime createdAt
        datetime updatedAt
    }

        int id PK
        string nombre

        int id_creador FK

        datetime updatedAt
    }

        int id PK

        datetime updatedAt
    }

    grupo_mensajes {
        int id PK
        int id_grupo FK
        int id_usuario FK
        text contenido
        datetime createdAt
        datetime updatedAt
    }

    mensaje_privados {
        int id PK
        int id_remitente FK
        int id_destinatario FK
        text contenido
        boolean leido
        datetime createdAt
        datetime updatedAt
    }

    foro_publicaciones {
        int id PK
        int id_materia FK
        int id_usuario FK
        string titulo
        text contenido
        string categoria
        int votos
        datetime createdAt
        datetime updatedAt
        datetime deletedAt
    }

    foro_comentarios {
        int id PK
        int id_publicacion FK
        int id_usuario FK
        text contenido
        int votos
        int id_comentario_padre FK
        datetime createdAt
        datetime updatedAt
    }

        int id PK

        string tipo
        datetime createdAt
        datetime updatedAt
    }

    foro_etiquetas {
        int id PK
        int id_publicacion FK
        string nombre
        datetime createdAt
        datetime updatedAt
    }

    foro_publicaciones_guardadas {
        int id PK
        int id_usuario FK
        int id_publicacion FK
        datetime createdAt
        datetime updatedAt
    }

    foro_reportes {
        int id PK
        int id_usuario_reportador FK
        int id_publicacion FK
        int id_comentario FK
        text descripcion
        boolean resuelto
        datetime createdAt
        datetime updatedAt
    }

    materiales_estudio {
        int id PK
        text ubicacion
        int id_materia FK
        string titulo
        text etiquetas
        int id_usuario FK
        datetime fecha_de_publicacion
        int likes
        int descargas
    }

    material_calificaciones {
        int id PK
        int id_material FK
        int id_usuario FK
        int puntuacion
        datetime createdAt
        datetime updatedAt
    }
